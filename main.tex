    \begin{tikzpicture}

            fill=#1,

            \draw[red!30, thin, step=1] (current page.south west) grid (current page.north east);

\newcommand{\drawdebugguides}[1]{
    \begin{tikzpicture}[remember picture, overlay]
        % 1. BLEED & TRIM ZONE (Red dashed line at the physical edge)
        \draw[red, thick, dashed] (current page.south west) rectangle (current page.north east);
        
        % 2. THE GUTTER (Center fold line)
        \draw[blue, line width=0.5pt] (current page.north) -- (current page.south);
        %\node[blue, anchor=north, font=\tiny\sffamily] at (current page.north) {FOLD / GUTTER};

        % 3. SAFETY MARGINS (10mm from every edge)
        % Any text outside this box is at risk of being trimmed or lost in the fold.
        \draw[orange, thin] ([shift={(10mm,10mm)}]current page.south west) 
                 rectangle ([shift={(-10mm,-10mm)}]current page.north east);

        % 4. COORDINATE GRID
        \draw[gray!20, thin, step=1] (current page.south west) grid (current page.north east);
        
        % 5. LABEL
        \node[fill=black, opacity=0.6, text=white, font=\ttfamily\scriptsize, anchor=north west] 
            at (current page.north west) {#1 | PAGE: \thepage};
    \end{tikzpicture}
}

% Usage: \singlepage{image}{x}{y}{width}{content}
\newcommand{\singlepage}[5]{
    \newpage
    \begin{tikzpicture}[remember picture, overlay]
        \node[inner sep=0] at (current page.center) 
            {\includegraphics[width=\paperwidth, height=\paperheight]{#1}};
    \end{tikzpicture}
    \drawdebugguides{SINGLE PAGE}
    \begin{textblock}{#4}(#2,#3)
        #5
    \end{textblock}
}

\newcommand{\textpage}[1]{
    \newpage
    \thispagestyle{empty}
    \drawdebugguides{TEXT PAGE}
    \begin{textblock}{6}(1,1)
        #1
    \end{textblock}
    \mbox{} % Keeps the page from being discarded if it looks "empty" to LaTeX
}

% Define the parity check first
\newcommand{\checkparity}{
    \ifodd\value{page}
%        \errmessage{Logic Error: Spread started on an ODD page (\thepage). This will break your booklet!}
        \begin{tikzpicture}[remember picture, overlay]
            \fill[red!10] (current page.south west) rectangle (current page.north east);
            \draw[red!30, thin, step=1] (current page.south west) grid (current page.north east);
            \node[red, font=\Huge\bfseries, rotate=45] at (current page.center) {WILL PRINT BADLY};
        \end{tikzpicture}
    \fi
}

% Usage: \prosespread[L/R]{image}{Text Content}
\newcommand{\prosespread}[3][L]{
    \clearpage
    \checkparity
    
    \if#1L
        % IMAGE ON LEFT, TEXT ON RIGHT
        \singlepage{#2}{0}{0}{0}{} % Empty textblock call for image page
        \clearpage
        \drawdebugguides{PROSE RIGHT}
        \begin{textblock}{6}(1,1)
            #3
        \end{textblock}
    \else
        % TEXT ON LEFT, IMAGE ON RIGHT
        \drawdebugguides{PROSE LEFT}
        \begin{textblock}{6}(1,1)
            #3
        \end{textblock}
        \clearpage
        \singlepage{#2}{0}{0}{0}{}
    \fi
}
